extends Node

var current_mana: int = 150

var turrets_db = {
    "turret_basic": {
        "name": "Arcane Sentry",
        "mana_cost": 50,
        "fire_rate": 1.0,
        "range": 15.0,
        "damage": 25,
        "element": "PHYSICAL"
    },
    "turret_fire": {
        "name": "Pyre Mortar",
        "mana_cost": 100,
        "fire_rate": 2.2,
        "range": 18.0,
        "damage": 60,
        "element": "FIRE"
    },
    "turret_freeze": {
        "name": "Frost Obelisk",
        "mana_cost": 120,
        "fire_rate": 1.5,
        "range": 12.0,
        "damage": 30,
        "element": "ICE"
    }
}

func try_build_turret(turret_id: String, node_pos: Vector3) -> bool:
    if not turrets_db.has(turret_id):
        return false
    var data = turrets_db[turret_id]
    if current_mana < data["mana_cost"]:
        return false
    
    current_mana -= data["mana_cost"]
    return true

func add_mana(amount: int) -> void:
    current_mana += amount
